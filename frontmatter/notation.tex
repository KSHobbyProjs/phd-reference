\section*{Linear Algebra}
\begin{description}
    \item[$\ket{\psi}$] Ket vector.
    \item[$\bra{\psi}$] Bra vector under the Hermitian inner product.
    \item[$|\psi)$] Ket vector when working with the c-product.
    \item[$(\psi|$] Bra vector when working with the c-product.
\end{description}
